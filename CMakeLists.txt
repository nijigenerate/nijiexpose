cmake_minimum_required(VERSION 3.24)
include(cmake/Version.cmake)
unset(NIJIEXPOSE_VERSION CACHE) # Retire the old fixed-version cache option.
nijiexpose_git_version("${CMAKE_CURRENT_SOURCE_DIR}")
project(nijiexpose_native VERSION ${NIJIEXPOSE_NUMERIC_VERSION} LANGUAGES C CXX)
message(STATUS "nijiexpose version: ${NIJIEXPOSE_VERSION}")
string(REPLACE "\\" "\\\\" NIJIEXPOSE_VERSION_CPP "${NIJIEXPOSE_VERSION}")
string(REPLACE "\"" "\\\"" NIJIEXPOSE_VERSION_CPP "${NIJIEXPOSE_VERSION_CPP}")
configure_file(native/src/version.hpp.in generated/version.hpp @ONLY)
configure_file(cmake/version.txt.in nijiexpose-version.txt @ONLY)
string(REPLACE "&" "&amp;" NIJIEXPOSE_VERSION_PLIST "${NIJIEXPOSE_VERSION}")
string(REPLACE "<" "&lt;" NIJIEXPOSE_VERSION_PLIST "${NIJIEXPOSE_VERSION_PLIST}")
string(REPLACE ">" "&gt;" NIJIEXPOSE_VERSION_PLIST "${NIJIEXPOSE_VERSION_PLIST}")
if(NOT WIN32 AND NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native application currently supports Windows, macOS and Linux.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(NIJIEXPOSE_RENDERER "OPENGL" CACHE STRING "Application renderer: VULKAN or OPENGL")
set_property(CACHE NIJIEXPOSE_RENDERER PROPERTY STRINGS VULKAN OPENGL)
if(NIJIEXPOSE_RENDERER STREQUAL "VULKAN")
    find_package(Vulkan REQUIRED COMPONENTS glslc)
    add_compile_definitions(NIJIEXPOSE_VULKAN=1)
elseif(NOT NIJIEXPOSE_RENDERER STREQUAL "OPENGL")
    message(FATAL_ERROR "Unknown NIJIEXPOSE_RENDERER=${NIJIEXPOSE_RENDERER}")
endif()

if(APPLE)
    set(BUILD_TESTING OFF CACHE BOOL "Build native comparison and test targets")
endif()
include(CTest)
option(NIJIEXPOSE_GPU_TESTS "Register tests requiring a working desktop OpenGL context" OFF)
set(NIJIEXPOSE_NICXLIVE_SOURCE_DIR "" CACHE PATH "Optional local nicxlive checkout (otherwise use pinned source)")

include(cmake/Vendor.cmake)
include(cmake/Dependencies.cmake)
if(APPLE)
    set(NIJIEXPOSE_LICENSE_DESTINATION "nijiexpose.app/Contents/Resources/licenses")
else()
    set(NIJIEXPOSE_LICENSE_DESTINATION "share/licenses/nijiexpose")
endif()
add_subdirectory(native)
include(cmake/Packaging.cmake)
