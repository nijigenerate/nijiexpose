file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/sdk.sha256" sdk_expected LIMIT_COUNT 1)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/sdk-windows-x64.zip" sdk_actual)
if(NOT sdk_actual STREQUAL sdk_expected)
    message(FATAL_ERROR "Bundled nijitrack SDK checksum mismatch")
endif()
set(bundled_sdk "${CMAKE_CURRENT_BINARY_DIR}/sdk-${sdk_actual}")
if(NOT EXISTS "${bundled_sdk}/mediapipe_tracking.lib")
    file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/sdk-windows-x64.zip" DESTINATION "${bundled_sdk}")
endif()
set(NIJIEXPOSE_TRACKING_SDK "${bundled_sdk}")
foreach(archive mediapipe_tracking.lib nijitrack_warp.lib)
    if(NOT EXISTS "${NIJIEXPOSE_TRACKING_SDK}/${archive}")
        message(FATAL_ERROR "nijitrack native SDK missing: ${NIJIEXPOSE_TRACKING_SDK}/${archive}")
    endif()
endforeach()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/models/models.lock.json" model_lock)
foreach(name face_landmarker.task hand_landmarker.task)
    string(JSON expected GET "${model_lock}" "${name}" sha256)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/models/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Tracker task checksum mismatch: ${name}")
    endif()
endforeach()
